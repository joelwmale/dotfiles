# ghboard Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `ghboard`, a PHP-TUI terminal dashboard showing GitHub CI/Actions, Dependabot alerts (by severity), open PRs, and open issues for all repos in a directory (or the current repo).

**Architecture:** Self-contained Composer project at `cli/ghboard/` with PSR-4 `src/`. Data flows from `GithubClient` (shells out to `gh` CLI) into immutable `RepoData` DTOs, held in `App` state, then rendered by `Dashboard` into a PHP-TUI scrollable table. A 100ms tick loop drives keyboard events and refresh scheduling.

**Tech Stack:** PHP 8.2+, php-tui/php-tui, Pest 2, gh CLI

```php
<?php

declare(strict_types=1);

namespace Ghboard;

final class Config
{
    private function __construct(
        public array $ignored,
        public int $refreshInterval,
        public string $baseDir,
        private readonly string $path,
    ) {}

    public static function load(?string $path = null): self
    {
        $path ??= self::defaultPath();

        if (!file_exists($path)) {
            $config = new self(
                ignored: [],
                refreshInterval: 60,
                baseDir: (getenv('HOME') ?: '') . '/Code',
                path: $path,
            );
            $config->persist();
            return $config;
        }

        $data = json_decode((string) file_get_contents($path), associative: true);

        return new self(
            ignored: $data['ignored'] ?? [],
            refreshInterval: $data['refresh_interval'] ?? 60,
            baseDir: $data['base_dir'] ?? (getenv('HOME') ?: '') . '/Code',
            path: $path,
        );
    }

    public function ignore(string $repoName): void
    {
        if (!in_array($repoName, $this->ignored, strict: true)) {
            $this->ignored[] = $repoName;
        }
        $this->persist();
    }

    private function persist(): void
    {
        $tmp = $this->path . '.tmp';
        file_put_contents($tmp, json_encode([
            'ignored' => $this->ignored,
            'refresh_interval' => $this->refreshInterval,
            'base_dir' => $this->baseDir,
        ], JSON_PRETTY_PRINT));
        rename($tmp, $this->path);
    }

    private static function defaultPath(): string
    {
        return (getenv('HOME') ?: '') . '/.ghboard.json';
    }
}
```

```php
<?php

declare(strict_types=1);

namespace Ghboard;

final class WorkflowRun
{
    public function __construct(
        public readonly string $name,
        public readonly string $status,
        public readonly ?string $conclusion,
    ) {}

    public function isPassed(): bool
    {
        return $this->status === 'completed'
            && in_array($this->conclusion, ['success', 'skipped', 'neutral'], strict: true);
    }

    public function isRunning(): bool
    {
        return in_array($this->status, ['in_progress', 'queued', 'requested', 'waiting'], strict: true);
    }

    public function isFailed(): bool
    {
        return $this->status === 'completed' && !$this->isPassed();
    }

    public function icon(): string
    {
        return match (true) {
            $this->isRunning() => '⟳',
            $this->isFailed()  => '✗',
            default            => '✓',
        };
    }
}
```

```php
<?php

declare(strict_types=1);

namespace Ghboard;

final class RepoDiscovery
{
    public function __construct(
        private readonly string $cwd,
        private readonly string $baseDir = '',
    ) {}

    public function isSingleRepoMode(): bool
    {
        return is_dir($this->cwd . '/.git');
    }

    /** @return RepoData[] */
    public function discover(array $ignored): array
    {
        if ($this->isSingleRepoMode()) {
            $repo = $this->parseRepo($this->cwd);
            return $repo !== null ? [$repo] : [];
        }

        $scanDir = $this->baseDir ?: $this->cwd;
        $repos = [];

        foreach (scandir($scanDir) as $entry) {
            if ($entry === '.' || $entry === '..') {
                continue;
            }

            $path = $scanDir . '/' . $entry;

            if (!is_dir($path) || !is_dir($path . '/.git')) {
                continue;
            }

            if (in_array($entry, $ignored, strict: true)) {
                continue;
            }

            $repo = $this->parseRepo($path);
            if ($repo !== null) {
                $repos[] = $repo;
            }
        }

        usort($repos, fn(RepoData $a, RepoData $b) => strcmp($a->name, $b->name));

        return $repos;
    }

    private function parseRepo(string $path): ?RepoData
    {
        $configFile = $path . '/.git/config';

        if (!file_exists($configFile)) {
            return null;
        }

        $content = (string) file_get_contents($configFile);
        $url = $this->extractRemoteUrl($content);

        if ($url === null) {
            return null;
        }

        [$owner, $name] = $this->parseGithubUrl($url);

        if ($owner === null || $name === null) {
            return null;
        }

        return new RepoData(name: $name, owner: $owner, path: $path);
    }

    private function extractRemoteUrl(string $gitConfig): ?string
    {
        if (preg_match('/\[remote "origin"\][^\[]*url\s*=\s*(.+)/s', $gitConfig, $matches)) {
            return trim(explode("\n", $matches[1])[0]);
        }
        return null;
    }

    /** @return array{string|null, string|null} */
    private function parseGithubUrl(string $url): array
    {
        // SSH: git@github.com:owner/repo.git
        // HTTPS: https://github.com/owner/repo.git
        if (preg_match('/github\.com[:\\/]([^\/]+)\/([^\/\s]+?)(?:\.git)?$/', $url, $m)) {
            return [$m[1], $m[2]];
        }
        return [null, null];
    }
}
```

```php
<?php

declare(strict_types=1);

namespace Ghboard;

final class GithubClient
{
    /** @var callable(string): string */
    private $executor;

    public function __construct(?callable $executor = null)
    {
        $this->executor = $executor ?? static function (string $command): string {
            $output = shell_exec($command);
            return $output !== null ? trim($output) : '[]';
        };
    }

    /** @return WorkflowRun[] */
    public function getWorkflowRuns(string $owner, string $repo): array
    {
        $output = ($this->executor)(
            "gh run list --repo {$owner}/{$repo} --limit 30 --json name,status,conclusion 2>/dev/null"
        );

        $data = json_decode($output, associative: true);

        if (!is_array($data)) {
            return [];
        }

        $seen = [];
        $runs = [];

        foreach ($data as $row) {
            $name = $row['name'] ?? '';
            if (isset($seen[$name])) {
                continue;
            }
            $seen[$name] = true;
            $runs[] = new WorkflowRun(
                name: $name,
                status: $row['status'] ?? '',
                conclusion: ($row['conclusion'] ?? null) ?: null,
            );
        }

        return $runs;
    }

    public function getDependabotSummary(string $owner, string $repo): DependabotSummary
    {
        $output = ($this->executor)(
            "gh api repos/{$owner}/{$repo}/dependabot/alerts?state=open&per_page=100 2>/dev/null"
        );

        $data = json_decode($output, associative: true);

        if (!is_array($data)) {
            return DependabotSummary::empty();
        }

        $critical = $high = $medium = 0;

        foreach ($data as $alert) {
            match ($alert['security_vulnerability']['severity'] ?? '') {
                'critical' => $critical++,
                'high'     => $high++,
                'medium'   => $medium++,
                default    => null,
            };
        }

        return new DependabotSummary($critical, $high, $medium);
    }

    public function getOpenPrCount(string $owner, string $repo): int
    {
        $output = ($this->executor)(
            "gh pr list --repo {$owner}/{$repo} --state open --json number 2>/dev/null"
        );

        $data = json_decode($output, associative: true);
        return is_array($data) ? count($data) : 0;
    }

    public function getOpenIssueCount(string $owner, string $repo): int
    {
        $output = ($this->executor)(
            "gh issue list --repo {$owner}/{$repo} --state open --json number 2>/dev/null"
        );

        $data = json_decode($output, associative: true);
        return is_array($data) ? count($data) : 0;
    }
}
```

```php
declare(strict_types=1);

namespace Ghboard;

use PhpTui\Tui\Extension\Core\Widget\ParagraphWidget;
use PhpTui\Tui\Extension\Core\Widget\Table\TableCell;
use PhpTui\Tui\Extension\Core\Widget\Table\TableRow;
use PhpTui\Tui\Extension\Core\Widget\Table\TableState;
use PhpTui\Tui\Extension\Core\Widget\TableWidget;
use PhpTui\Tui\Layout\Constraint;
use PhpTui\Tui\Style\Style;
use PhpTui\Tui\Color\AnsiColor;
use PhpTui\Tui\Text\Line;
use PhpTui\Tui\Text\Span;
use PhpTui\Tui\Text\Text;

final class Dashboard
{
    public function buildHeader(int $repoCount, int $secondsSinceRefresh, bool $refreshing): ParagraphWidget
    {
        $status = $refreshing
            ? Span::styled(' refreshing...', Style::default()->fg(AnsiColor::Yellow))
            : Span::styled(" last refresh: {$secondsSinceRefresh}s ago", Style::default()->fg(AnsiColor::DarkGray));

        return ParagraphWidget::fromText(Text::fromLine(Line::fromSpans([
            Span::styled('ghboard', Style::default()->fg(AnsiColor::Blue)),
            Span::styled("  ·  {$repoCount} repos", Style::default()->fg(AnsiColor::DarkGray)),
            $status,
            Span::styled(
                '  ·  [r] refresh  [i] ignore  [enter] open  [q] quit',
                Style::default()->fg(AnsiColor::DarkGray)
            ),
        ])));
    }

    /**
     * @param RepoData[] $repos
     */
    public function buildTable(array $repos, TableState $tableState): TableWidget
    {
        $rows = array_map(fn(RepoData $repo) => $this->buildRow($repo), $repos);

        return TableWidget::default()
            ->header(
                TableRow::fromCells([
                    TableCell::fromString('REPO'),
                    TableCell::fromString('WORKFLOWS'),
                    TableCell::fromString('DEPENDABOT'),
                    TableCell::fromString('PRS'),
                    TableCell::fromString('ISSUES'),
                ])->style(Style::default()->fg(AnsiColor::DarkGray))
            )
            ->rows($rows)
            ->widths([
                Constraint::percentage(20),
                Constraint::percentage(34),
                Constraint::percentage(24),
                Constraint::percentage(11),
                Constraint::percentage(11),
            ])
            ->highlightStyle(Style::default()->bg(AnsiColor::DarkGray));
    }

    public function buildFooter(int $selectedIndex, int $total, int $secondsUntilRefresh): ParagraphWidget
    {
        return ParagraphWidget::fromText(Text::fromLine(Line::fromSpans([
            Span::styled(
                " ↕ j/k to scroll  ·  {$selectedIndex}/{$total}  ·  auto-refresh in {$secondsUntilRefresh}s",
                Style::default()->fg(AnsiColor::DarkGray)
            ),
        ])));
    }

    private function buildRow(RepoData $repo): TableRow
    {
        $nameSpan = Span::styled($repo->name, Style::default()->fg(AnsiColor::White));

        if ($repo->loading) {
            return TableRow::fromCells([
                TableCell::fromLine(Line::fromSpan($nameSpan)),
                TableCell::fromLine(Line::fromSpan(
                    Span::styled('loading...', Style::default()->fg(AnsiColor::DarkGray))
                )),
                TableCell::fromString(''),
                TableCell::fromString(''),
                TableCell::fromString(''),
            ]);
        }

        if ($repo->error !== null) {
            return TableRow::fromCells([
                TableCell::fromLine(Line::fromSpan($nameSpan)),
                TableCell::fromLine(Line::fromSpan(
                    Span::styled('error fetching data', Style::default()->fg(AnsiColor::Red))
                )),
                TableCell::fromString(''),
                TableCell::fromString(''),
                TableCell::fromString(''),
            ]);
        }

        return TableRow::fromCells([
            TableCell::fromLine(Line::fromSpan($nameSpan)),
            TableCell::fromLine($this->buildWorkflowsLine($repo->workflows)),
            TableCell::fromLine($this->buildDependabotLine($repo->dependabot)),
            TableCell::fromLine($this->buildCountLine($repo->openPrs)),
            TableCell::fromLine($this->buildCountLine($repo->openIssues)),
        ]);
    }

    /** @param WorkflowRun[] $workflows */
    private function buildWorkflowsLine(array $workflows): Line
    {
        if (empty($workflows)) {
            return Line::fromSpan(Span::styled('—', Style::default()->fg(AnsiColor::DarkGray)));
        }

        $spans = [];
        foreach ($workflows as $i => $workflow) {
            if ($i > 0) {
                $spans[] = Span::raw('  ');
            }

            $color = match (true) {
                $workflow->isRunning() => AnsiColor::Yellow,
                $workflow->isFailed()  => AnsiColor::Red,
                default                => AnsiColor::Green,
            };

            $spans[] = Span::styled(
                $workflow->icon() . ' ' . $workflow->name,
                Style::default()->fg($color)
            );
        }

        return Line::fromSpans($spans);
    }

    private function buildDependabotLine(DependabotSummary $dependabot): Line
    {
        if (!$dependabot->hasAlerts()) {
            return Line::fromSpan(Span::styled('—', Style::default()->fg(AnsiColor::DarkGray)));
        }

        $spans = [];

        if ($dependabot->critical > 0) {
            $spans[] = Span::styled("● {$dependabot->critical} crit", Style::default()->fg(AnsiColor::Red));
        }
        if ($dependabot->high > 0) {
            if (!empty($spans)) {
                $spans[] = Span::raw('  ');
            }
            $spans[] = Span::styled("● {$dependabot->high} high", Style::default()->fg(AnsiColor::Yellow));
        }
        if ($dependabot->medium > 0) {
            if (!empty($spans)) {
                $spans[] = Span::raw('  ');
            }
            $spans[] = Span::styled("● {$dependabot->medium} med", Style::default()->fg(AnsiColor::DarkGray));
        }

        return Line::fromSpans($spans);
    }

    private function buildCountLine(int $count): Line
    {
        $color = $count > 0 ? AnsiColor::White : AnsiColor::DarkGray;
        return Line::fromSpan(Span::styled((string) $count, Style::default()->fg($color)));
    }
}
```

```php
<?php

declare(strict_types=1);

namespace Ghboard;

use PhpTui\Term\Terminal;
use PhpTui\Term\Event\CharKeyEvent;
use PhpTui\Term\Event\CtrlCEvent;
use PhpTui\Tui\DisplayBuilder;
use PhpTui\Tui\Extension\Core\Widget\Table\TableState;
use PhpTui\Tui\Layout\Constraint;
use PhpTui\Tui\Layout\Layout;
use PhpTui\Tui\Direction;

final class App
{
    private bool $quit = false;
    private bool $forceRefresh = false;
    private bool $refreshing = false;
    private int $lastRefreshedAt = 0;

    /** @var RepoData[] */
    private array $repos = [];

    private TableState $tableState;
    private Dashboard $dashboard;

    public function __construct(
        private readonly Config $config,
        private readonly ?GithubClient $client = null,
        private readonly ?RepoDiscovery $discovery = null,
    ) {
        $this->tableState = new TableState();
        $this->dashboard = new Dashboard();
    }

    public function run(): void
    {
        $terminal = Terminal::new();
        $terminal->enableRawMode();

        register_shutdown_function(static function () use ($terminal): void {
            $terminal->disableRawMode();
        });

        $display = DisplayBuilder::default($terminal)->build();
        $client = $this->client ?? new GithubClient();
        $discovery = $this->discovery ?? new RepoDiscovery(
            cwd: (string) getcwd(),
            baseDir: str_replace('~', (string) (getenv('HOME') ?: ''), $this->config->baseDir),
        );

        $this->repos = $discovery->discover(ignored: $this->config->ignored);
        $this->tableState->select(0);

        // Load data immediately on first run
        $this->fetchAll($client);

        while (!$this->quit) {
            foreach ($terminal->events()->drain() as $event) {
                $this->handleEvent($event, $client);
            }

            $elapsed = time() - $this->lastRefreshedAt;
            if (!$this->refreshing && ($elapsed >= $this->config->refreshInterval || $this->forceRefresh)) {
                $this->forceRefresh = false;
                $this->fetchAll($client);
            }

            $secondsSince = time() - $this->lastRefreshedAt;
            $secondsUntil = max(0, $this->config->refreshInterval - $secondsSince);
            $repoCount = count($this->repos);
            $selectedIndex = ($this->tableState->selected() ?? 0) + 1;
            $refreshing = $this->refreshing;
            $repos = $this->repos;
            $tableState = $this->tableState;

            $display->draw(function ($frame) use (
                $secondsSince, $secondsUntil, $repoCount, $selectedIndex, $refreshing, $repos, $tableState
            ): void {
                $area = $frame->area();

                $chunks = Layout::default()
                    ->direction(Direction::Vertical)
                    ->constraints([
                        Constraint::length(1),
                        Constraint::min(0),
                        Constraint::length(1),
                    ])
                    ->split($area);

                $frame->renderWidget(
                    $this->dashboard->buildHeader($repoCount, $secondsSince, $refreshing),
                    $chunks[0]
                );

                $frame->renderStatefulWidget(
                    $this->dashboard->buildTable($repos, $tableState),
                    $chunks[1],
                    $tableState
                );

                $frame->renderWidget(
                    $this->dashboard->buildFooter($selectedIndex, $repoCount, $secondsUntil),
                    $chunks[2]
                );
            });

            usleep(100_000); // 100ms tick
        }
    }

    private function handleEvent(mixed $event, GithubClient $client): void
    {
        if ($event instanceof CtrlCEvent) {
            $this->quit = true;
            return;
        }

        if (!$event instanceof CharKeyEvent) {
            return;
        }

        $selected = $this->tableState->selected() ?? 0;
        $count = count($this->repos);

        match ($event->char) {
            'q'  => $this->quit = true,
            'r'  => $this->forceRefresh = true,
            'j'  => $this->tableState->select(min($selected + 1, $count - 1)),
            'k'  => $this->tableState->select(max($selected - 1, 0)),
            'i'  => $this->ignoreSelected(),
            "\r" => $this->openInBrowser($selected),
            default => null,
        };
    }

    private function fetchAll(GithubClient $client): void
    {
        $this->refreshing = true;
        $this->lastRefreshedAt = time();

        foreach ($this->repos as $i => $repo) {
            try {
                $workflows  = $client->getWorkflowRuns($repo->owner, $repo->name);
                $dependabot = $client->getDependabotSummary($repo->owner, $repo->name);
                $prs        = $client->getOpenPrCount($repo->owner, $repo->name);
                $issues     = $client->getOpenIssueCount($repo->owner, $repo->name);

                $this->repos[$i] = $repo->withData($workflows, $dependabot, $prs, $issues);
            } catch (\Throwable $e) {
                $this->repos[$i] = $repo->withError($e->getMessage());
            }
        }

        $this->refreshing = false;
    }

    private function ignoreSelected(): void
    {
        $selected = $this->tableState->selected() ?? 0;

        if (!isset($this->repos[$selected])) {
            return;
        }

        $this->config->ignore($this->repos[$selected]->name);
        array_splice($this->repos, $selected, 1);
        $this->tableState->select(max(0, min($selected, count($this->repos) - 1)));
    }

    private function openInBrowser(int $index): void
    {
        if (!isset($this->repos[$index])) {
            return;
        }

        $repo = $this->repos[$index];
        shell_exec("gh repo view {$repo->owner}/{$repo->name} --web 2>/dev/null &");
    }
}
```
